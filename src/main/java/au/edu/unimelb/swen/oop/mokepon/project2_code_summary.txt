        MokeponChooseView fxMokeponChooseView = new FxMokeponChooseView();

        MokeponSelectView mokeponSelectView = new MokeponSelectView(
                fxMokeponListView, fxMokeponDetailView, fxMokeponChooseView
        );

        ScreenController mokeponSelectController = new MokeponSelectController(
                mokeponSelectView, gameState, mokeponList, eventBus, randomSource, screenManager
        );






        screenManager.register(ScreenState.SELECT, mokeponSelectController);




        mokeponSelectController.init();


        Group mainRoot = (Group) gameBgView.getBgViewRoot();
        mainRoot.getChildren().addAll(
                mokeponSelectView.getMokeponSelectRoot()
        );
        Scene scene = new Scene(mainRoot);
        primaryStage.setScene(scene);
        primaryStage.show();

    }

    private Properties loadPropertiesFromArgument() throws IOException {
        String argument = getParameters().getRaw().getFirst();
        if (!argument.startsWith("data=")) {
            throw new IllegalArgumentException("Missing expected argument in the form data=relativeFilePath");
        }

        Properties properties = new Properties();
        try (var input = Files.newInputStream(Path.of(argument.substring("data=".length())))) {
            properties.load(input);
        }
        return properties;
    }

    public static void main(String[] args) {
        Application.launch(args);
    }
}

import javafx.scene.Parent;
import javafx.scene.layout.Pane;
import javafx.stage.Screen;
import javafx.stage.Stage;

import java.util.HashMap;
import java.util.LinkedHashMap;

public class ScreenManager {

    private Stage stage;
    private HashMap<ScreenState, ScreenController> controllerMap = new LinkedHashMap<>();
    private ScreenState currentScreenState;

    public ScreenManager(Stage stage) {
        this.stage = stage;
        currentScreenState = null;
    }


    public void register(ScreenState screenState, ScreenController screenController) {
        this.controllerMap.put(screenState, screenController);
    }

    public void switchTo(ScreenState screenState) {
        if (currentScreenState != null) {
            controllerMap.get(currentScreenState).hide();
        }
        ScreenController targetController = controllerMap.get(screenState);
        if (targetController == null) return;

        targetController.show();
        currentScreenState = screenState;

        Parent currentRoot = stage.getScene().getRoot();
        //stage.getScene().setRoot(null); // add view.node()
    }

    public ScreenState getCurrentScreenState() {
        return currentScreenState;
    }
}

package au.edu.unimelb.swen.oop.mokepon.controller;

import au.edu.unimelb.swen.oop.mokepon.app.ScreenManager;
import au.edu.unimelb.swen.oop.mokepon.app.ScreenState;
import au.edu.unimelb.swen.oop.mokepon.event.ChooseMokeponEvent;
import au.edu.unimelb.swen.oop.mokepon.event.EventBus;
import au.edu.unimelb.swen.oop.mokepon.model.Mokepon;
import au.edu.unimelb.swen.oop.mokepon.state.GameState;
import au.edu.unimelb.swen.oop.mokepon.util.random.RandomSource;
import au.edu.unimelb.swen.oop.mokepon.view.MokeponSelectView;
import javafx.scene.Node;

import java.util.ArrayList;
import java.util.function.Consumer;

public class MokeponSelectController implements ScreenController {

    private MokeponSelectView view;
    private GameState state;
    private ArrayList<Mokepon> mokeponList;
    private EventBus eventBus;
    private RandomSource randomSource;
    private ScreenManager screenManager;


    public MokeponSelectController(
            MokeponSelectView view,
            GameState state,
            ArrayList<Mokepon> mokeponList,
            EventBus eventBus,
            RandomSource randomSource,
            ScreenManager screenManager) {
        this.view = view;
        this.state = state;
        this.mokeponList = mokeponList;
        this.eventBus = eventBus;
        this.randomSource = randomSource;
        this.screenManager = screenManager;
    }


    @Override
    public void init() {
        Consumer<Mokepon> btnClickEvent = mokepon -> {
            if ((state.getSelectedMokepon() == null) || !state.getSelectedMokepon().equals(mokepon)) {
                onMokeponSelected(mokepon);
            } else {
                onMokeponCancelSelected();
            }
        };
        view.getMokeponListView().setItems(mokeponList, btnClickEvent);
        state.selectedMokeponProperty().addListener((observable, oldValue, newValue) -> {
            if (newValue == null) {
                view.getMokeponListView().clearSelection();
                view.getMokeponDetailView().clear();
            }
        });
    }

    @Override
    public void show() {
        view.getMokeponSelectRoot().setVisible(true);
    }

    @Override
    public void hide() {
        view.getMokeponSelectRoot().setVisible(false);
    }

    @Override
    public Node getRoot() {
        return view.getMokeponSelectRoot();
    }

    private void onMokeponSelected(Mokepon mokepon) {
        state.setSelectedMokepon(mokepon);
        view.getMokeponListView().select(mokepon);
        view.getMokeponDetailView().show(mokepon);
    }

    private void onMokeponCancelSelected() {
        state.setSelectedMokepon(null);
        view.getMokeponDetailView().clear();
    }

    private void onChooseClicked() {
        Mokepon selectedMokepon = state.getSelectedMokepon();
        if (selectedMokepon == null) return;

        state.setPlayerMokepon(selectedMokepon);

        Mokepon enemyMokepon = mokeponList.get(randomSource.nextInt(0, mokeponList.size()));
        state.setEnemyMokepon(enemyMokepon);

        eventBus.publish(new ChooseMokeponEvent(selectedMokepon, enemyMokepon, state));
        screenManager.switchTo(ScreenState.BATTLE);
    }
}

package au.edu.unimelb.swen.oop.mokepon.battle;

import au.edu.unimelb.swen.oop.mokepon.battle.skill.Skill;
import au.edu.unimelb.swen.oop.mokepon.model.Mokepon;
import au.edu.unimelb.swen.oop.mokepon.util.random.RandomSource;

import java.util.ArrayList;
import java.util.HashMap;

public class BattleService {

    private RandomSource randomSource;

    private HashMap<String, Skill> skillMap = new HashMap<>();

    public BattleService(RandomSource randomSource) {
        this.randomSource = randomSource;
    }


    public BattleResult useSkill(Skill skill, Mokepon self, Mokepon other, BattleContext battleContext) {
        if (self.getCurrentEnergy() < skill.getEnergyCost()) {
            ArrayList<String> messageList = new ArrayList<>();
            messageList.add(self.getName() + " do not have enough energy for " + skill.getName() + "!");
            return new BattleResult(messageList, BattleOutcome.ONGOING);
        }

        self.setCurrentEnergy(self.getCurrentEnergy() - skill.getEnergyCost());
        BattleResult skillResult = skill.execute(self, other, battleContext);
        BattleOutcome outcome = checkBattleEnd(self, other);
        if (outcome != BattleOutcome.ONGOING) return new BattleResult(skillResult.getMessageList(), outcome);
        return skillResult;
    }

    public BattleResult useEnemyRandomSkill(Mokepon self, Mokepon other, BattleContext battleContext) {
        int randonSkillInt = randomSource.nextInt(0, self.getSkills().size());
        Skill skill = SkillRegistry.getSkill(self.getSkills().get(randonSkillInt));
        return useSkill(skill, self, other, battleContext);
    }

    public BattleOutcome checkBattleEnd(Mokepon self, Mokepon other) {
        if (other.getCurrentHP() <= 0) return BattleOutcome.PLAYER_WIN;
        if (self.getCurrentHP() <= 0) return BattleOutcome.PLAYER_LOSE;
        return BattleOutcome.ONGOING;
    }

}

package au.edu.unimelb.swen.oop.mokepon.event;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.function.Consumer;

public class EventBus {

    private HashMap<Class<?>, ArrayList<Consumer<?>>> subscriberListMap = new HashMap<>();

    public <T> void subscribe(Class<T> eventType, Consumer<T> handler) {
        subscriberListMap.computeIfAbsent(eventType, _ -> new ArrayList<>()).add(handler);
    }

    @SuppressWarnings("unchecked")
    public <T> void publish(T event) {
        ArrayList<Consumer<?>> handlerList = subscriberListMap.get(event.getClass());
        if (handlerList == null) return;

        handlerList.forEach(handler -> {
            ((Consumer<T>)handler).accept(event);
        });
    }

}

            itemMap.put(itemType, new SimpleIntegerProperty(0));
        }
    }


    public LinkedHashMap<ItemType, IntegerProperty> getItemMap() {
        return itemMap;
    }

    public void add(ItemType itemType) {
        IntegerProperty itemNum = itemMap.get(itemType);
        itemNum.setValue(itemNum.getValue() + 1);
    }

    public int getCount(ItemType itemType) {
        return itemMap.get(itemType).getValue();
    }

    public boolean use(ItemType itemType) {
        IntegerProperty itemNum = itemMap.get(itemType);
        if (itemNum.getValue() <= 0) return false;

        itemNum.setValue(itemNum.getValue() - 1);
        return true;
    }

    public void clear() {
        itemMap.values().forEach(item -> {
            item.setValue(0);
        });
    }

}

import java.util.HashMap;
import java.util.function.Function;

public class FontLoader {

    private static HashMap<Integer, Font> fontCacheMap = new HashMap<>();

    private final static String GOSTISCH_FONT_URL = "/assets/GrenzeGotisch-Black.ttf";


    public static Font getGostischFont(int size) {

        Function<Integer, Font> fontGenerator = fontSize -> {
            Font resultFont = new Font(fontSize);
            try (InputStream inputFile = FontLoader.class.getResourceAsStream(GOSTISCH_FONT_URL)) {
                resultFont = Font.loadFont(inputFile, fontSize);
            } catch (Exception e) {
                e.printStackTrace();
            } finally {
                return resultFont;
            }
        };

        return fontCacheMap.computeIfAbsent(size, fontGenerator);
    }

}

        menuItemArr[index].setOnAction(_ -> onClick.accept(index));

        scrollPane.skinProperty().addListener((observer, oldSkin, newSkin) -> {
            Node viewport = scrollPane.lookup(".viewport");
            if (viewport != null)
                viewport.setStyle("-fx-background-color: transparent;");

            Node scrollBar = scrollPane.lookup(".scroll-bar");
            if (scrollBar != null)
                scrollBar.setStyle("-fx-background-color: transparent;");
        });
        scrollPane.setPrefSize(300, 600);

        mokeponSelectTitle.setFont(titleFont);

        selectBlock.getChildren().addAll(mokeponSelectTitle, scrollPane);
        selectBlock.setAlignment(Pos.CENTER);
    }

    public Node node() {
        return selectBlock;
    }

    @Override
    public void setItems(ArrayList<Mokepon> mokeponList, Consumer<Mokepon> onSelect) {
        listBox.getChildren().clear();
        buttonMap.clear();
        for (Mokepon mokepon : mokeponList) {
            Button nameBtn = new Button(mokepon.getName());
            nameBtn.setFont(btnFont);
            nameBtn.setStyle("-fx-background-radius: 999px;" +
                    "-fx-background-color: #d3d3d3;" +
                    "-fx-text-fill: #000000;"
            );
            nameBtn.setOpacity(0.2);
            nameBtn.setOnMouseEntered(click -> {
                nameBtn.setOpacity(0.5);
            });
            nameBtn.setOnMouseExited(click -> {
                nameBtn.setOpacity(0.2);
            });
            nameBtn.setOnMouseClicked(click -> {
                nameBtn.setOpacity(0.75);
            });
            nameBtn.setPrefSize(280, 25);
            nameBtn.setOnAction(event -> onSelect.accept(mokepon));
            buttonMap.put(mokepon, nameBtn);
            listBox.getChildren().add(nameBtn);
        }
    }

    @Override
    public String getSelectedButtonName() {
        return selectedButton.getText();
    }

    @Override
    public void select(Mokepon mokepon) {
        clearSelection();
        selectedButton = buttonMap.get(mokepon);
    }

    @Override
    public void clearSelection() {
        selectedButton = null;
    }
}

        }
        battleItemRoot.getChildren().addAll(itemBtnMap.values());
        battleItemRoot.setAlignment(Pos.CENTER);
    }




    @Override
    public Node node() {
        return battleItemRoot;
    }

    @Override
    public void setItems(HashMap<ItemType, IntegerProperty> itemDeposit, Consumer<IntegerProperty> onUseItem) {
        itemDeposit.forEach((itemType, integerProperty) -> {
            Button itemBtn = itemBtnMap.get(itemType.name());
            itemBtn.setOnMouseEntered(enter -> {
                itemBtn.setOpacity(0.5);
            });
            itemBtn.setOnMouseExited(exit -> {
                itemBtn.setOpacity(0.2);
            });
            itemBtn.setOnMouseClicked(event -> {
                itemBtn.setOpacity(0.75);
                onUseItem.accept(integerProperty);
            });
            integerProperty.addListener((observable, oldValue, newValue) -> {
                if ((Integer)newValue <= 0) {
                    itemBtn.setDisable(true);
                } else {
                    itemBtn.setDisable(false);
                }
            });
        });
    }

    @Override
    public void setDisable(boolean disable) {
        itemBtnMap.values().forEach(itemBtn -> {
            itemBtn.setDisable(disable);
        });
    }
