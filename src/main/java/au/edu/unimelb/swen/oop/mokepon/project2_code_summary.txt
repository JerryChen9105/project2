ARCHITECTURAL SUMMARY (one paragraph):
The project has been refactored from a single monolithic Main class into a
layered MVC/MVP-style architecture with clear separation: model (Mokepon,
MokeponType), state (GameState with JavaFX properties, Inventory), data loaders
(GameDataLoader abstract base + MokeponLoader + TileLoader stub), a battle
subsystem (BattleService + Skill strategy classes), an event system (EventBus
topic-based pub/sub), view INTERFACES under view/view_interface.* with JavaFX
implementations under view/fx.*, screen FACADES in view.* (GameBgView,
MokeponSelectView, MokeponBattleView), controllers (ScreenController + concrete
controllers), and an app shell (GameWindow extends Stage, ScreenManager,
ScreenState enum). The SELECT screen is now largely wired and functional
(loader -> view -> controller -> state properties -> detail display). The BATTLE
and OVERWORLD screens plus battle logic remain stubbed/empty.

public class Main extends Application {

    private ScreenManager screenManager;
    private GameState gameState;
    private EventBus eventBus;
    private GameBgView gameBgView;

    @Override
    public void start(Stage primaryStage) throws Exception {

        primaryStage = new GameWindow();
        screenManager = new ScreenManager(primaryStage);
        gameState = new GameState();
        eventBus = new EventBus();
        gameBgView = new GameBgView(new FxBgView());

        Properties gameProperty = loadPropertiesFromArgument();

        MokeponLoader mokeponLoader = new MokeponLoader(gameProperty);
        ArrayList<Mokepon> mokeponList = new ArrayList<>(mokeponLoader.loadAll().values());

        MokeponListView fxMokeponListView = new FxMokeponListView();
        MokeponDetailView fxMokeponDetailView = new FxMokeponDetailView();

        MokeponSelectView mokeponSelectView = new MokeponSelectView(fxMokeponListView, fxMokeponDetailView);

        ScreenController mokeponSelectController = new MokeponSelectController(
                mokeponSelectView, gameState, mokeponList, eventBus
        );

        screenManager.register(ScreenState.SELECT, mokeponSelectController);

        mokeponSelectController.init();

        Group mainRoot = new Group();
        mainRoot.getChildren().addAll(gameBgView.getBgViewRoot(), mokeponSelectView.getMokeponSelectRoot());
        Scene scene = new Scene(mainRoot);
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private Properties loadPropertiesFromArgument() throws IOException {
        String argument = getParameters().getRaw().getFirst();
        if (!argument.startsWith("data=")) {
            throw new IllegalArgumentException("Missing expected argument in the form data=relativeFilePath");
        }

        Properties properties = new Properties();
        try (var input = Files.newInputStream(Path.of(argument.substring("data=".length())))) {
            properties.load(input);
        }
        return properties;
    }

    public static void main(String[] args) {
        Application.launch(args);
    }
}

OBSERVATIONS:

    public ScreenManager(Stage stage) {
        this.stage = stage;
        currentScreenState = null;
    }

    public void register(ScreenState screenState, ScreenController screenController) {
        this.controllerMap.put(screenState, screenController);
    }

    public void switchTo(ScreenState screenState) {
        if (currentScreenState != null) {
            controllerMap.get(currentScreenState).hide();
        }
        ScreenController targetController = controllerMap.get(screenState);
        if (targetController == null) return;

        targetController.show();
        currentScreenState = screenState;
        stage.getScene().setRoot(null); // add view.node()  <-- TODO / INCOMPLETE
    }

    public ScreenState getCurrentScreenState() {
        return currentScreenState;
    }
}

package au.edu.unimelb.swen.oop.mokepon.controller;

import au.edu.unimelb.swen.oop.mokepon.event.EventBus;
import au.edu.unimelb.swen.oop.mokepon.model.Mokepon;

import java.util.ArrayList;
import java.util.function.Consumer;

public class MokeponSelectController implements ScreenController {

    private MokeponSelectView view;
    private GameState state;
    private ArrayList<Mokepon> mokeponList;
    private EventBus eventBus;

    public MokeponSelectController(
            MokeponSelectView view,
            GameState state,
            ArrayList<Mokepon> mokeponList,
            EventBus eventBus) {
        this.view = view;
        this.state = state;
        this.mokeponList = mokeponList;
        this.eventBus = eventBus;
    }

    @Override
    public void init() {
        Consumer<Mokepon> btnClickEvent = mokepon -> {
            if ((state.getSelectedMokepon() == null) || !state.getSelectedMokepon().equals(mokepon)) {
                onMokeponSelected(mokepon);
            } else {
                onChooseClicked();
            }
        };
        view.getMokeponListView().setItems(mokeponList, btnClickEvent);
        state.selectedMokeponProperty().addListener((observable, oldValue, newValue) -> {
            if (newValue == null) {
                view.getMokeponListView().clearSelection();
                view.getMokeponDetailView().clear();
            }
        });
    }

    @Override
    public void show() {
        view.getMokeponSelectRoot().setVisible(true);
    }

    @Override
    public void hide() {
        view.getMokeponSelectRoot().setVisible(false);
    }

    private void onMokeponSelected(Mokepon mokepon) {
        state.setSelectedMokepon(mokepon);
        view.getMokeponDetailView().show(mokepon);
    }

    private void onChooseClicked() {
        state.setSelectedMokepon(null);
        view.getMokeponDetailView().clear();
    }
}

OBSERVATIONS:
- init() wires the list view with a callback and subscribes to selectedMokepon
  changes.
- The click callback implements a toggle: first click selects, second click on
  the SAME already-selected mokepon triggers onChooseClicked().
- BUG/LOGIC: onChooseClicked() sets selectedMokepon to null and clears the
  detail view, i.e. it behaves like a DESELECT rather than a confirm/choose.
  The intended "choose -> go to battle" behaviour (increment fightCount, set
  playerMokepon, switchTo BATTLE) is NOT implemented.
- onMokeponSelected() does not call view.getMokeponListView().select(mokepon),
  so the list view's internal selectedButton is never set (getSelectedButtonName
  would still NPE).

package au.edu.unimelb.swen.oop.mokepon.state;

import javafx.beans.property.IntegerProperty;
import javafx.beans.property.SimpleIntegerProperty;
import java.util.HashMap;

public class Inventory {

    private HashMap<ItemType, IntegerProperty> itemMap = new HashMap<>();

    public Inventory() {
        for (ItemType itemType : ItemType.values()) {
            itemMap.put(itemType, new SimpleIntegerProperty(0));
        }
    }

    public void add(ItemType itemType) {
        IntegerProperty itemNum = itemMap.get(itemType);
        itemNum.setValue(itemNum.getValue() + 1);
    }

    public int getCount(ItemType itemType) {
        return itemMap.get(itemType).getValue();
    }

    public boolean use(ItemType itemType) {
        IntegerProperty itemNum = itemMap.get(itemType);
        if (itemNum.getValue() <= 0) return false;
        itemNum.setValue(itemNum.getValue() - 1);
        return true;
    }

    public void clear() {
        itemMap.values().forEach(item -> {
            item.setValue(0);
        });
    }
}
// Fully implemented. Effect of using an item (heal/poison/coffee) is NOT applied
// here - use() only decrements the count.

             + "\nmaxEnergy : " + maxEnergy + "\nbaseStrength : " + baseStrength
             + "\nbaseAccuracy : " + baseAccuracy + "\nbaseDefence : " + baseDefence
             + "\nbaseDodge : " + baseDodge + "\nskills : " + skills;
    }

    public boolean equals(Mokepon other) {
        boolean skillsAreSame = true;
        for (int i = 0; i < this.skillList.size(); i++) {
            if (!this.skillList.get(i).equals(other.skillList.get(i))) {
                skillsAreSame = false;
                break;
            }
        }
        return this.name.equals(other.getName()) &&      // now uses .equals (fixed)
               this.imgURL.equals(other.getImgURL()) &&
               skillsAreSame;
    }
    // NOTE: overloads equals WITHOUT @Override / hashCode -> violates equals/hashCode
    // contract; equality also ignores maxHP/energy/stats/type.

    // base-stat getters
    public String getName() { return this.name; }
    public String getImgURL() { return this.imgURL; }
    public int getMaxHP() { return this.maxHP; }
    public int getMaxEnergy() { return this.maxEnergy; }
    public int getBaseStrength() { return this.baseStrength; }
    public int getBaseAccuracy() { return this.baseAccuracy; }
    public int getBaseDefence() { return this.baseDefence; }
    public int getBaseDodge() { return this.baseDodge; }
    public ArrayList<String> getSkills() { return this.skillList; }
    public MokeponType getMokeponType() { return mokeponType; }

    // current-stat getters
    public int getCurrentHP() { return this.currentHP; }
    public int getCurrentEnergy() { return this.currentEnergy; }
    public int getCurrentStrength() { return this.currentStrength; }
    public int getCurrentAccuracy() { return this.currentAccuracy; }
    public int getCurrentDefence() { return this.currentDefence; }
    public int getCurrentDodge() { return currentDodge; }

    // setters (clamped)
    public void setCurrentHP(int hp) { if (hp < 0) hp = 0; this.currentHP = hp; }
    public void setCurrentEnergy(int energy) { if (energy < 0) energy = 0; this.currentEnergy = energy; }
    public void setCurrentDefence(int defence) {
        if (defence < 0) defence = 0;
        else if (defence > 100) defence = 100;
        this.currentDefence = defence;    // FIXED (previously wrote to currentEnergy)
    }

    public void takeDamage(int damage) {
        if (this.currentHP <= damage) this.currentHP = 0;
        else this.currentHP -= damage;
    }
    public void healToMax() { this.currentHP = this.maxHP; }
    public void restoreEnergyToMax() { this.currentEnergy = this.maxEnergy; }
    public void resetBattleStats() {
        healToMax(); restoreEnergyToMax();
        this.setCurrentDefence(this.baseDefence);
    }

    public Mokepon copy() {
        return new Mokepon(
                this.name, this.imgURL, this.maxHP, this.maxEnergy,
                this.baseStrength, this.baseAccuracy, this.baseDefence, this.baseDodge,
                new ArrayList<>(this.skillList), this.mokeponType   // now deep-copies skillList
        );
    }
}

package au.edu.unimelb.swen.oop.mokepon.event;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.function.Consumer;

public class EventBus {

    private HashMap<Class<?>, ArrayList<Consumer<?>>> subscriberListMap = new HashMap<>();

    public <T> void subscribe(Class<T> eventType, Consumer<T> handler) {
        subscriberListMap.computeIfAbsent(eventType, _ -> new ArrayList<>()).add(handler);
    }

    @SuppressWarnings("unchecked")
    public <T> void publish(T event) {
        ArrayList<Consumer<?>> handlerList = subscriberListMap.get(event.getClass());
        if (handlerList == null) return;

        handlerList.forEach(handler -> {
            ((Consumer<T>)handler).accept(event);
        });
    }
}
// Uses the unnamed variable '_' in the lambda (Java 21+). Routes by exact runtime
// class of the event; subclasses of an event type will not match a supertype
// subscription.

--- util/FontLoader.java (font loading with size-keyed cache) ---
package au.edu.unimelb.swen.oop.mokepon.util;

import javafx.scene.text.Font;
import java.io.InputStream;
import java.util.HashMap;
import java.util.function.Function;

public class FontLoader {

    private static HashMap<Integer, Font> fontCacheMap = new HashMap<>();
    private final static String GOSTISCH_FONT_URL = "/assets/GrenzeGotisch-Black.ttf";

    public static Font getGostischFont(int size) {
        Function<Integer, Font> fontGenerator = fontSize -> {
            Font resultFont = new Font(fontSize);
            try (InputStream inputFile = FontLoader.class.getResourceAsStream(GOSTISCH_FONT_URL)) {
                resultFont = Font.loadFont(inputFile, fontSize);
            } catch (Exception e) {
                e.printStackTrace();
            } finally {
                return resultFont;
            }
        };
        return fontCacheMap.computeIfAbsent(size, fontGenerator);
    }
}
// NOTE: uses computeIfAbsent (cleaner than before). The 'finally { return ... }'
// inside the lambda is unusual (try/finally returning a value). Method name has
// typo "Gostisch". Class has no private constructor guard; cache is static.

public class ImageLoader {

    public static Image load(String url, double width, double height) {

        try {
            return new Image(url, width, height, true, false);
        } catch (IllegalArgumentException e) {
            return null;
        }
    }

    public static Image load(String url, double width, double height, boolean preserveRatio) {
        if (url == null || url.isBlank()) return null;
        try {
            return new Image(url, width, height, preserveRatio, false);
        } catch (IllegalArgumentException e) {
            return null;
        }
    }
}

// swallow IllegalArgumentException and return null.

--- data/MokeponLoader.java (reads all mokepon.N.* keys) ---
package au.edu.unimelb.swen.oop.mokepon.data;

import au.edu.unimelb.swen.oop.mokepon.model.Mokepon;
import au.edu.unimelb.swen.oop.mokepon.model.MokeponType;
import java.util.*;

public class MokeponLoader extends GameDataLoader<String, Mokepon> {

    private LinkedHashMap<String, Mokepon> mokeponMap = new LinkedHashMap<>();
    private final static int MAXMOKEPONNUM = 50;

    public MokeponLoader(Properties gameData) {
        super(gameData);
        String propertyPrefix;
        int mokeponCount = 0;
        while (mokeponCount < MAXMOKEPONNUM) {
            propertyPrefix = "mokepon." + mokeponCount + ".";
            try {
                String name = gameData.getProperty(propertyPrefix + "name");
                String imgURL = gameData.getProperty(propertyPrefix + "image");
                String hp = gameData.getProperty(propertyPrefix + "hp");
                String energy = gameData.getProperty(propertyPrefix + "energy");
                String strength = gameData.getProperty(propertyPrefix + "strength");
                String accuracy = gameData.getProperty(propertyPrefix + "accuracy");
                String defence = gameData.getProperty(propertyPrefix + "defence");
                String dodge = gameData.getProperty(propertyPrefix + "dodge");
                String skillStr = gameData.getProperty(propertyPrefix + "skills");
                String type = gameData.getProperty(propertyPrefix + "type");
                ArrayList<String> skillStrList = new ArrayList<>();
                Collections.addAll(skillStrList, skillStr.split("\\|"));

                mokeponMap.put(name, new Mokepon(name, imgURL, Integer.parseInt(hp), Integer.parseInt(energy),
                        Integer.parseInt(strength), Integer.parseInt(accuracy), Integer.parseInt(defence),
                        Integer.parseInt(dodge), skillStrList, MokeponType.valueOf(type)));
                mokeponCount++;
            } catch (Exception e) {
                break;
            }
        }
    }

    @Override public Mokepon load(String obj) { return mokeponMap.get(obj); }
    @Override public LinkedHashMap<String, Mokepon> loadAll() { return mokeponMap; }
}
// NOTE: parsing loop stops at first exception (relies on contiguous indices).
// Reads `type` and maps to MokeponType.valueOf(...). Argument order is now
// (strength, accuracy, defence, dodge) matching the Mokepon constructor
// (baseDefence, baseDodge) - the previous dodge/defence swap appears FIXED.
// Swallows per-entry parse errors silently (break).

The mokepon.game file is a Java .properties file (547 lines). Keys observed
