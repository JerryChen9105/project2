package au.edu.unimelb.swen.oop.mokepon.app;

import au.edu.unimelb.swen.oop.mokepon.controller.ScreenController;
import javafx.stage.Stage;

import java.util.HashMap;
import java.util.LinkedHashMap;

public class ScreenManager {

    private Stage stage;
    private HashMap<ScreenState, ScreenController> controllerMap = new LinkedHashMap<>();
    private ScreenState currentScreenState;

    public ScreenManager(Stage stage) {
        this.stage = stage;
        currentScreenState = null;
    }


    public void register(ScreenState screenState, ScreenController screenController) {
        this.controllerMap.put(screenState, screenController);
    }

    public void switchTo(ScreenState screenState) {
        ScreenController targetController = controllerMap.get(screenState);
        if (targetController == null) return;

        if (currentScreenState != null) controllerMap.get(currentScreenState).hide();
        targetController.init();
        targetController.show();
        currentScreenState = screenState;
    }

    public void terminateGame() {
        stage.close();
    }

    public ScreenState getCurrentScreenState() {
        return currentScreenState;
    }
}

package au.edu.unimelb.swen.oop.mokepon.controller;

import au.edu.unimelb.swen.oop.mokepon.app.ScreenManager;

import java.util.ArrayList;

public class MokeponBattleController implements ScreenController {

    private MokeponBattleView view;
    private GameState state;
    private BattleService battleService;
    private EventBus eventBus;
    private BattleContext context = new BattleContext();
    private RandomSource randomSource;
    private ScreenManager screenManager;


    public MokeponBattleController(MokeponBattleView view,
                                   GameState state,
                                   BattleService battleService,
                                   EventBus eventBus,
                                   BattleContext context,
                                   RandomSource randomSource,
                                   ScreenManager screenManager) {
        this.view = view;
        this.state = state;
        this.battleService = battleService;
        this.eventBus = eventBus;
        this.context = context;
        this.randomSource = randomSource;
        this.screenManager = screenManager;
    }


    @Override
    public void init() {
        Mokepon self = state.getPlayerMokepon();
        Mokepon enemy = state.getEnemyMokepon();
        if (self == null || enemy == null) return;

        self.setMokeponStatus();
        enemy.setMokeponStatus();
        view.getSelfView().initSelfStatus(self, this::onSkillClicked,
                state.getInventory().getItemMap(), this::onItemUsed);
        view.getEnemyView().initEnemyStatus(enemy);
        view.getLogView().clear();
        view.getLogView().append("Player chose " + self.getName() +
                ". Began a fight against enemy " + enemy.getName() + "!\n\n");
    }

    @Override
    public void show() {
        view.getMokeponBattleRoot().setVisible(true);
    }

    @Override
    public void hide() {
        view.getMokeponBattleRoot().setVisible(false);
    }

    @Override
    public Node getRoot() {
        return view.getMokeponBattleRoot();
    }

    private void onSkillClicked(Skill skill) {
        if (state.getPlayerMokepon().getMokeponStatus().getCurrentEnergy() < skill.getEnergyCost()) return;

        BattleResult battleResult = skill.execute(state.getPlayerMokepon(),
                state.getEnemyMokepon(),
                context);
        battleResult.getMessageList().forEach(message -> {
            view.getLogView().append(message);
        });

        view.getSelfView().updateSelfStatus(state.getPlayerMokepon());
        view.getEnemyView().updateEnemyStatus(state.getEnemyMokepon());

        if (state.getEnemyMokepon().getMokeponStatus().getCurrentHP() <= 0) {
            handleBattleEnd(BattleOutcome.PLAYER_WIN);
            return;
        }

        runEnemyTurn();
    }

    private void onItemUsed(ItemType itemType) {
        state.getInventory().use(itemType);

        // 1. 鐗╁搧鍙兘浼氬奖鍝嶇帺瀹剁姸鎬侊紝闇€瑕佸皢鍏跺簲鐢ㄥ埌 Mokepon 涓?(鍏蜂綋閫昏緫鍙栧喅浜庝綘鐨?Item 璁捐)

        view.getSelfView().updateSelfStatus(state.getPlayerMokepon());
        view.getEnemyView().updateEnemyStatus(state.getEnemyMokepon());
    }

    private void runEnemyTurn() {
        Mokepon enemy = state.getEnemyMokepon();
        Mokepon player = state.getPlayerMokepon();

        ArrayList<Skill> availableSkillList = new ArrayList<>();
        enemy.getMokeponStatus().getSkillList().forEach(skill -> {
            if (skill.getEnergyCost() <= enemy.getMokeponStatus().getCurrentEnergy()) {
                availableSkillList.add(skill);
            }
        });

        Skill chosenSkill = availableSkillList.get(randomSource.nextInt(0, availableSkillList.size()));

        BattleResult result = chosenSkill.execute(enemy, player, context);
        result.getMessageList().forEach(message -> {
            view.getLogView().append(message);
        });

        view.getSelfView().updateSelfStatus(player);
        view.getEnemyView().updateEnemyStatus(enemy);

        if (player.getMokeponStatus().getCurrentHP() <= 0) {
            handleBattleEnd(BattleOutcome.PLAYER_LOSE);
        }
    }

    private void handleBattleEnd(BattleOutcome battleOutcome) {
        // 绂佺敤鐜╁鎿嶄綔锛岄槻姝㈠湪鎴樻枟缁撶畻鏈熼棿缁х画鐐瑰嚮鎶€鑳?
        // view.getSelfView().disableInteraction();

        switch (battleOutcome) {
            case PLAYER_WIN -> {
                view.getLogView().append("You won the battle!\n");
                // 绛夊緟鍚庣画寮€鍙?WinView 鍚庡彇娑堟敞閲?
                // screenManager.switchTo("PLAYER_WIN_SCREEN");
            }
            case PLAYER_LOSE -> {
                view.getLogView().append("You lost the battle...\n");
                // 绛夊緟鍚庣画寮€鍙?LossView 鍚庡彇娑堟敞閲?
                // screenManager.switchTo("PLAYER_LOSS_SCREEN");
            }
            default -> {
                view.getLogView().append("\n");
            }
        }

        // 鍙€夛細浣跨敤 EventBus 骞挎挱鎴樻枟缁撴潫浜嬩欢锛屾柟渚垮叾浠栫郴缁?濡備换鍔＄郴缁熴€佹垚灏辩郴缁?鐩戝惉
        // eventBus.post(new BattleEndedEvent(battleOutcome));

import java.util.ArrayList;

public class MokeponOverworldController implements ScreenController {


    private MokeponOverworldView view;
    private GameState gameState;
    private EventBus eventBus;
    private RandomSource randomSource;
    private ScreenManager screenManager;

    private ArrayList<ArrayList<Character>> tileMatrix;
    private OverworldSystem overworldSystem;


    public MokeponOverworldController(MokeponOverworldView view,
                                      GameState gameState,
                                      EventBus eventBus,
                                      RandomSource randomSource,
                                      ScreenManager screenManager) {
        this.view = view;
        this.gameState = gameState;
        this.eventBus = eventBus;
        this.randomSource = randomSource;
        this.screenManager = screenManager;

        TileLoader tileLoader = new TileLoader(GameData.getGameData());
        this.tileMatrix = new ArrayList<>(tileLoader.loadAll().values());

        initOverworldSys();
        view.getMovementView().setUpAction(this::onUpClicked);
        view.getMovementView().setDownAction(this::onDownClicked);
        view.getMovementView().setLeftAction(this::onLeftClicked);
        view.getMovementView().setRightAction(this::onRightClicked);
    }


    @Override
    public void init() {
        initOverworldSys();

        view.getTileView().clear();
        view.getTileView().setTile(tileMatrix);
        view.getTileView().putPlayer(PlayerInfoLoader.getStartingCoordinate());

        EnemyInfoLoader.getCoordinateEnemyMap().forEach((coordinate, enemyType) -> {
            view.getTileView().putEntity(coordinate, EnemyInfoLoader.getEnemyImageView(enemyType));
        });
        ItemInfoLoader.getCoordinateItemMap().forEach((coordinate, itemType) -> {
            view.getTileView().putEntity(coordinate, ItemInfoLoader.getItemImgView());
        });
        view.getTileView().putEntity(ExitInfoLoader.getExitCoordinate(), ExitInfoLoader.getExitImgView());


        view.getPlayerStatisticView().setMokeponStatistic(gameState.selectedMokeponProperty());
        view.getPlayerStatisticView().setInventoryStatistic(gameState.getInventory());
        view.getPlayerStatisticView().setFightCount(gameState.fightCountProperty());


        view.getStatusBarView().clear();


        view.getExitMarkerView().setReturnButton(this::onReturnClicked);
        view.getExitMarkerView().setQuitButton(this::onQuitClicked);
        view.getExitMarkerView().setVisible(false);
    }

    private void initOverworldSys() {
        Coordinate playerCoordinate = PlayerInfoLoader.getStartingCoordinate();
        Coordinate exitCoordinate = ExitInfoLoader.getExitCoordinate();

        this.overworldSystem = new OverworldSystem(tileMatrix,
                playerCoordinate, exitCoordinate,
                EnemyInfoLoader.getCoordinateEnemyMap(),
                ItemInfoLoader.getCoordinateItemMap());
    }

    private void handleMovement(int dx, int dy) {
        // 1. 鍚庣寮曟搸杩愯浆
        OverworldStatus overworldStatus = overworldSystem.movePlayer(dx, dy);

        // 2. 鍙湁鍚堟硶绉诲姩锛孶I 鎵嶄細鏇存柊
        if (playerMoveControl(overworldStatus)) {
            // 鐩存帴鍚戠郴缁熻缁濆鍑嗙‘鐨勬渶鏂板潗鏍囷紒
            Coordinate realPos = overworldSystem.getPlayerCoordinate();
            // 鐢变簬 FxTileView 閲?putPlayer 浣跨敤鐨勬槸 GridPane 淇敼 Index锛屼笉鐢ㄦ媴蹇冩畫褰遍棶棰?
            view.getTileView().putPlayer(realPos);
        }
    }

    private void onUpClicked() {
        handleMovement(0, -1);
    }
    private void onDownClicked() {
        handleMovement(0, 1);
    }
    private void onLeftClicked() {
        handleMovement(-1, 0);
    }
    private void onRightClicked() {
        handleMovement(1, 0);
    }

    private boolean playerMoveControl(OverworldStatus overworldStatus) {
        view.getStatusBarView().clear();
        switch (overworldStatus) {
            case STOP -> {
                view.getStatusBarView().setStatusText("Illegal path to go!");
                return false; // 鎾炲锛屼腑鏂搷浣?
            }
            case MOVE -> {
                return true; // 绌哄湴锛屽厑璁哥Щ鍔?
            }
            case ITEM -> {
                ItemType itemType = overworldSystem.checkOutItemType();
                if (itemType != ItemType.UNKNOWN) {
                    gameState.getInventory().add(itemType); // 瀛樺叆鑳屽寘
                    view.getPlayerStatisticView().setInventoryStatistic(gameState.getInventory()); // 鍒锋柊渚ц竟鏍?
                    view.getStatusBarView().setStatusText("You picked up a " + itemType.name() + "!");

                    // 閫氱煡 UI 浠庡湴鍥句笂鎿﹂櫎閭ｄ釜瀹濈
                    view.getTileView().popEntity(overworldSystem.getPlayerCoordinate());
                }
                return true; // 鍏佽绉诲姩
            }
            case ENEMY -> {
                // 鈿狅笍 鏋佸叾鍏抽敭锛佽繖閲屽繀椤?return true 鍏佽鐜╁璧板埌杩欎竴鏍硷紝
                // 鐒跺悗鎵嶈兘瑙﹀彂閬囨晫鏈哄埗銆傜敱浜庝綘杩樻病鍐欐垬鏂楁ā鍧楋紝鍏堝脊涓彁绀恒€?
                System.out.println("Encountered enemy at: " + overworldSystem.getPlayerX() + ", " + overworldSystem.getPlayerY());
                view.getStatusBarView().setStatusText("Encountered a wild Mokepon!");

                // TODO: 缁撳悎闅惧害鍜岄殢鏈烘暟鐢熸垚鏁屾柟绮剧伒锛屽苟寮瑰嚭 view.getMokeponMarkerView()

                return true;
            }
            case EXIT -> {
                view.getExitMarkerView().setVisible(true);
                return true; // 鍏佽韪╁埌闂ㄤ笂
            }
            default -> {
                return false;
            }
        }
    }

    @Override
    public void show() {
        view.getMokeponOverworldRoot().setVisible(true);
    }

    @Override
    public void hide() {
        view.getMokeponOverworldRoot().setVisible(false);
    }

    @Override
    public Node getRoot() {
        return view.getMokeponOverworldRoot();
    }



    private void onQuitClicked() {
        screenManager.terminateGame();
    }

    private void onReturnClicked() {
        gameState.resetForNewSelection();
        eventBus.publish(new ExitOverworldEvent(gameState));
        screenManager.switchTo(ScreenState.SELECT);
    }

}

package au.edu.unimelb.swen.oop.mokepon.controller;

import javafx.scene.Node;

import java.util.ArrayList;
import java.util.function.Consumer;

public class MokeponSelectController implements ScreenController {

    private MokeponSelectView view;
    private GameState state;
    private EventBus eventBus;
    private ScreenManager screenManager;

    private ArrayList<Mokepon> mokeponList;


    public MokeponSelectController(
            MokeponSelectView view,
            GameState state,
            EventBus eventBus,
            ScreenManager screenManager) {
        this.view = view;
        this.state = state;
        this.eventBus = eventBus;
        this.screenManager = screenManager;

        MokeponLoader mokeponLoader = new MokeponLoader(GameData.getGameData());
        this.mokeponList = new ArrayList<>(mokeponLoader.loadAll().values());
    }


    @Override
    public void init() {
        state.resetForNewSelection();
        Consumer<Mokepon> btnClickEvent = mokepon -> {
            if ((state.getSelectedMokepon() == null) || !state.getSelectedMokepon().equals(mokepon)) {
                onMokeponSelected(mokepon);
            } else {
                onMokeponCancelSelected();
            }
        };
        view.getMokeponListView().setItems(mokeponList, btnClickEvent);

        state.selectedMokeponProperty().addListener((observable, oldValue, newValue) -> {
            if (newValue == null) {
                view.getMokeponListView().clearSelection();
                view.getMokeponDetailView().clear();
            }
        });

        view.getMokeponChooseView().setChooseAction(this::onChooseClicked);
    }

    @Override
    public void show() {
        view.getMokeponSelectRoot().setVisible(true);
    }

    @Override
    public void hide() {
        view.getMokeponSelectRoot().setVisible(false);
    }

    @Override
    public Node getRoot() {
        return view.getMokeponSelectRoot();
    }

    private void onMokeponSelected(Mokepon mokepon) {
        state.setSelectedMokepon(mokepon);
        view.getMokeponListView().select(mokepon);
        view.getMokeponDetailView().show(mokepon);
    }

    private void onMokeponCancelSelected() {
        state.setSelectedMokepon(null);
        view.getMokeponDetailView().clear();
    }

    private void onChooseClicked() {
        Mokepon selectedMokepon = state.getSelectedMokepon();
        if (selectedMokepon == null) return;

        state.setPlayerMokepon(selectedMokepon);

        eventBus.publish(new ChooseMokeponEvent(selectedMokepon, state));
        screenManager.switchTo(ScreenState.OVERWORLD);
    }
}

package au.edu.unimelb.swen.oop.mokepon.data;

import au.edu.unimelb.swen.oop.mokepon.driven.model.Mokepon;
import au.edu.unimelb.swen.oop.mokepon.driven.model.MokeponType;

import java.util.*;

public class MokeponLoader extends GameDataLoader<String, Mokepon> {

    private LinkedHashMap<String, Mokepon> mokeponMap = new LinkedHashMap<>();

    private final static int MAXMOKEPONNUM = 50;


    public MokeponLoader(Properties gameData) {
        super(gameData);
        String propertyPrefix;
        int mokeponCount = 0;
        while (mokeponCount < MAXMOKEPONNUM) {
            propertyPrefix = "mokepon." + mokeponCount + ".";
            try {
                String name = gameData.getProperty(propertyPrefix + "name");
                String imgURL = gameData.getProperty(propertyPrefix + "image");
                String hp = gameData.getProperty(propertyPrefix + "hp");
                String energy = gameData.getProperty(propertyPrefix + "energy");
                String strength = gameData.getProperty(propertyPrefix + "strength");
                String accuracy = gameData.getProperty(propertyPrefix + "accuracy");
                String defence = gameData.getProperty(propertyPrefix + "defence");
                String dodge = gameData.getProperty(propertyPrefix + "dodge");
                String skillStr = gameData.getProperty(propertyPrefix + "skills");
                String type = gameData.getProperty(propertyPrefix + "type");
                ArrayList<String> skillStrList = new ArrayList<>();
                Collections.addAll(skillStrList, skillStr.split("\\|"));

                mokeponMap.put(name, new Mokepon(name, imgURL, Integer.parseInt(hp), Integer.parseInt(energy),
                        Integer.parseInt(strength), Integer.parseInt(accuracy), Integer.parseInt(defence),
                        Integer.parseInt(dodge), skillStrList, MokeponType.valueOf(type)));
                mokeponCount++;
            } catch (Exception e) {
                break;
            }
        }
    }

    @Override
    public Mokepon load(String obj) {
        return mokeponMap.get(obj);
    }

    public LinkedHashMap<String, Mokepon> loadAll() {
        return mokeponMap;
    }

}

package au.edu.unimelb.swen.oop.mokepon.data;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.Properties;

public class TileLoader extends GameDataLoader<Integer, ArrayList<Character>> {


    private LinkedHashMap<Integer, ArrayList<Character>> tileMap = new LinkedHashMap<>();

    private static final int MAXTILECOUNT = 20;


    public TileLoader(Properties gameData) {
        super(gameData);
        String propertyPrefix;
        int tileCount = 0;
        while (tileCount < MAXTILECOUNT) {
            propertyPrefix = "map.tiles." + tileCount;
            try {
                String tileStr = gameData.getProperty(propertyPrefix);
                ArrayList<Character> charList = new ArrayList<>();
                for (char c : tileStr.toCharArray()) {
                    charList.add(c);
                }
                tileMap.put(tileCount, charList);
                tileCount++;
            } catch (Exception e) {
                break;
            }

package au.edu.unimelb.swen.oop.mokepon.driven.battle;

import au.edu.unimelb.swen.oop.mokepon.driven.model.Mokepon;

import au.edu.unimelb.swen.oop.mokepon.util.random.RandomSource;

import java.util.ArrayList;
import java.util.HashMap;

public class BattleService {

    private RandomSource randomSource;

    private HashMap<String, Skill> skillMap = new HashMap<>();

    public BattleService(RandomSource randomSource) {
        this.randomSource = randomSource;
    }


    public BattleResult useSkill(Skill skill, Mokepon self, Mokepon other, BattleContext battleContext) {
        if (self.getMokeponStatus().getCurrentEnergy() < skill.getEnergyCost()) {
            ArrayList<String> messageList = new ArrayList<>();
            messageList.add(self.getName() + " do not have enough energy for " + skill.getName() + "!");
            return new BattleResult(messageList, BattleOutcome.ONGOING);
        }

        self.getMokeponStatus().setCurrentEnergy(self.getMokeponStatus().getCurrentEnergy() - skill.getEnergyCost());
        BattleResult skillResult = skill.execute(self, other, battleContext);
        BattleOutcome outcome = checkBattleEnd(self, other);
        if (outcome != BattleOutcome.ONGOING) return new BattleResult(skillResult.getMessageList(), outcome);
        return skillResult;
    }

    public BattleResult useEnemyRandomSkill(Mokepon self, Mokepon other, BattleContext battleContext) {
        ArrayList<String> affordableSkillList = new ArrayList<>();
        int currentEnergy = self.getMokeponStatus().getCurrentEnergy();

        for (String skillName : self.getSkills()) {
            Skill skill = SkillRegistry.getSkill(skillName);
            if (skill.getEnergyCost() <= currentEnergy) {
                affordableSkillList.add(skillName);
            }
        }

        int randomIndex = randomSource.nextInt(0, affordableSkillList.size());
        Skill chosenSkill = SkillRegistry.getSkill(affordableSkillList.get(randomIndex));

        return useSkill(chosenSkill, self, other, battleContext);
    }

    public BattleOutcome resolveTurnEnd(Mokepon player, Mokepon enemy, BattleContext context) {
        // 闇€姹傦細鍏堢粨绠楁晫浜虹殑鐘舵€?
        Effect[] enemyExpiredEffectArr = enemy.getMokeponStatus().updateEffect();
        // 锛堣繖閲岄渶瑕佺‘淇濇偍鐨?Effect 鍦ㄨЕ鍙?onTurnEnd 鏃讹紝濡傛灉閫犳垚浜嗕激瀹筹紝瑕佸啓杩?context.log 涓級
        if (enemy.getMokeponStatus().getCurrentHP() <= 0) {
            return BattleOutcome.PLAYER_WIN;
        }

        // 鏁屼汉娌℃锛屽啀缁撶畻鐜╁鐨勭姸鎬?
        Effect[] playerExpiredEffectArr = player.getMokeponStatus().updateEffect();
        if (player.getMokeponStatus().getCurrentHP() <= 0) {
            return BattleOutcome.PLAYER_LOSE;
        }

        return BattleOutcome.ONGOING;
    }


    public BattleOutcome checkBattleEnd(Mokepon self, Mokepon other) {
        if (other.getMokeponStatus().getCurrentHP() <= 0) return BattleOutcome.PLAYER_WIN;
        if (self.getMokeponStatus().getCurrentHP() <= 0) return BattleOutcome.PLAYER_LOSE;
        return BattleOutcome.ONGOING;
    }

public class BleedEffect implements Effect {

    private int remainingTurn;
    private int userStrength;


    public BleedEffect(int userStrength) {
        this.remainingTurn = 2;
        this.userStrength = userStrength;
    }

    @Override
    public void onTurnEnd(MokeponStatus owner, BattleContext battleContext) {
        if (remainingTurn > 0) {
            int bleedDamage = (int) (0.5 * userStrength * 50 / owner.getCurrentDefence());
            owner.takeDamage(bleedDamage);
            remainingTurn--;
        }
        if (isExpired()) battleContext.log("The bleeding stopped.");
    }

    @Override
    public boolean isExpired() {
        return remainingTurn <= 0;
    }

    @Override
    public String getName() {
        return "Bleed";
    }
}

package au.edu.unimelb.swen.oop.mokepon.driven.model;

import au.edu.unimelb.swen.oop.mokepon.driven.model.skill.SkillRegistry;
import au.edu.unimelb.swen.oop.mokepon.driven.model.effect.Effect;
import au.edu.unimelb.swen.oop.mokepon.driven.model.skill.Skill;

import java.util.ArrayList;
import java.util.Iterator;

public class MokeponStatus {

    private int currentHP;
    private int currentEnergy;
    private int currentStrength;
    private int currentAccuracy;
    private int currentDefence;
    private int currentDodge;
    private ArrayList<Skill> skillList;
    private ArrayList<Effect> effectList;


    public MokeponStatus(int currentHP, int currentEnergy,
                         int currentStrength, int currentAccuracy,
                         int currentDefence, int currentDodge,
                         ArrayList<String> skillList) {
        this.currentHP = currentHP;
        this.currentEnergy = currentEnergy;
        this.currentStrength = currentStrength;
        this.currentAccuracy = currentAccuracy;
        this.currentDefence = currentDefence;
        this.currentDodge = currentDodge;
        this.skillList = new ArrayList<>();
        skillList.forEach(skillStr -> {
            this.skillList.add(SkillRegistry.getSkill(skillStr));
        });
        this.effectList = new ArrayList<>();
    }


    public int getCurrentHP() {
        return currentHP;
    }

    public int getCurrentEnergy() {
        return currentEnergy;
    }

    public int getCurrentStrength() {
        return currentStrength;
    }

    public int getCurrentAccuracy() {
        return currentAccuracy;
    }

    public int getCurrentDefence() {
        return currentDefence;
    }

    public int getCurrentDodge() {
        return currentDodge;
    }

    public ArrayList<Skill> getSkillList() {
        return skillList;
    }

    public ArrayList<Effect> getEffectList() {
        return effectList;
    }

    public void setCurrentHP(int hp) {
        currentHP = hp;
    }

    public void setCurrentEnergy(int energy) {
        currentEnergy = energy;
    }

    public void setCurrentStrength(int strength) {
        currentStrength = strength;
    }

    public void setCurrentAccuracy(int accuracy) {
        currentAccuracy = accuracy;
    }

    public void setCurrentDefence(int defence) {
        currentDefence = defence;
    }

    public void setCurrentDodge(int dodge) {
        currentDodge = dodge;
    }

    public void takeDamage(int damage) {
        if (currentHP <= damage) currentHP = 0;
        else currentHP -= damage;
    }

    public void addEffect(Effect effect) {
        effectList.add(effect);
    }

    public boolean hasEffect(String effectName) {
        for (Effect effect : effectList) {
            if (effect.getName().equals(effectName) && !effect.isExpired()) return true;
        }
        return false;
    }

    public Effect[] updateEffect() {
        if (effectList.isEmpty()) return null;

        ArrayList<Effect> expiredEffects = new ArrayList<>();
        Iterator<Effect> iterator = effectList.iterator();
        while (iterator.hasNext()) {
            Effect effect = iterator.next();
            effect.onTurnEnd(this);
            if (effect.isExpired()) {
                expiredEffects.add(effect);
                iterator.remove();
            }
        }
        if (expiredEffects.isEmpty()) return null;
        return expiredEffects.toArray(new Effect[0]);
    }

}

        ArrayList<String> messageList = new ArrayList<>();
        messageList.add("Player's " + user.getName() + " used Swole!\n");

            user.getMokeponStatus().setCurrentStrength(100);

        return new BattleResult(messageList, BattleOutcome.ONGOING);

package au.edu.unimelb.swen.oop.mokepon.driven.model.skill.util;

import au.edu.unimelb.swen.oop.mokepon.driven.model.MokeponStatus;
import au.edu.unimelb.swen.oop.mokepon.util.random.JavaRandomSource;
import au.edu.unimelb.swen.oop.mokepon.util.random.RandomSource;

public class HitCalculator {

    private static final RandomSource randomSource = new JavaRandomSource();

    public static boolean check(MokeponStatus selfStatus, MokeponStatus otherStatus) {
        double hitProb = Math.pow(selfStatus.getCurrentAccuracy(), 2) /
                (Math.pow(selfStatus.getCurrentAccuracy(), 2) + Math.pow(otherStatus.getCurrentDodge(), 2));

        boolean baseHit = hitProb >= randomSource.nextDouble(0.0, 1.0);

        if (baseHit && selfStatus.hasEffect("Blind")) {
            return randomSource.nextDouble(0.0, 1.0) >= 0.5;
        } // add else if branch

        return baseHit;
    }


}

package au.edu.unimelb.swen.oop.mokepon.driven.overworld;

public class Coordinate {

    private int x, y;


    public Coordinate(int x, int y) {
        this.x = x;
        this.y = y;
    }


    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Coordinate that = (Coordinate) o;
        return x == that.x && y == that.y;
    }

    @Override
    public int hashCode() {
        return java.util.Objects.hash(x, y);
    }


    public int getX() {
        return x;
    }

    public int getY() {
        return y;
    }

    public void setX(int nx) {
        x = nx;
    }
    public void setY(int ny) {
        y = ny;
    }

    public void setXY(int nx, int ny) {
        this.setX(nx);
        this.setY(ny);
    }

    public void setXY(Coordinate coordinate) {
        this.setXY(coordinate.getX(), coordinate.getY());
    }

}

package au.edu.unimelb.swen.oop.mokepon.driven.overworld;

import au.edu.unimelb.swen.oop.mokepon.state.Inventory;
import au.edu.unimelb.swen.oop.mokepon.state.ItemType;

import java.util.ArrayList;
import java.util.LinkedHashMap;

public class OverworldSystem {

    private WorldMap worldMap;
    private Coordinate player;
    private Coordinate lastCoordinate;

    private LinkedHashMap<Coordinate, OverworldStatus> coordinateStatusMap;
    private LinkedHashMap<Coordinate, ItemType> coordinateItemMap;
    private LinkedHashMap<Coordinate, EnemyType> coordinateEnemyMap;

    private Inventory inventory;


    public OverworldSystem(ArrayList<ArrayList<Character>> worldMapListMatrix,
                           Coordinate playerCoordinate,
                           Coordinate exitCoordinate,
                           LinkedHashMap<Coordinate, EnemyType> coordinateEnemyMap,
                           LinkedHashMap<Coordinate, ItemType> coordinateItemMap) {
        worldMap = new WorldMap(worldMapListMatrix);
        player = new Coordinate(playerCoordinate.getX(), playerCoordinate.getY());
        lastCoordinate = new Coordinate(player.getX(), player.getY());

        this.coordinateEnemyMap = coordinateEnemyMap;
        this.coordinateItemMap = coordinateItemMap;

        coordinateStatusMap = new LinkedHashMap<>();
        coordinateEnemyMap.forEach((coordinate, enemyType) -> {
            coordinateStatusMap.put(coordinate, OverworldStatus.ENEMY);
        });
        coordinateItemMap.forEach((coordinate, status) -> {
            coordinateStatusMap.put(coordinate, OverworldStatus.ITEM);
        });
        coordinateStatusMap.put(exitCoordinate, OverworldStatus.EXIT);
    }

    public int getPlayerX() { return player.getX(); }
    public int getPlayerY() { return player.getY(); }
    public Coordinate getPlayerCoordinate() { return new Coordinate(player.getX(), player.getY()); }

    public ItemType checkOutItemType() {
        ItemType itemType = coordinateItemMap.get(player);
        if (itemType == null) return ItemType.UNKNOWN;

        coordinateItemMap.remove(player);
        coordinateStatusMap.remove(player);
        return itemType;
    }

    public OverworldStatus movePlayer(int dx, int dy) {
        int newX = player.getX() + dx;
        int newY = player.getY() + dy;
        Coordinate targetCoord = new Coordinate(newX, newY);

        TileType tileType = worldMap.getTileType(targetCoord);
        if (!tileType.isWalkable()) {
            return OverworldStatus.STOP;
        }

        lastCoordinate.setXY(player);
        player.setXY(newX, newY);

        OverworldStatus status = coordinateStatusMap.get(player);
        if (status != null) {
            return status;
        }

        return OverworldStatus.MOVE;
    }

    public void revertMove() {
        player.setXY(lastCoordinate);
    }

}

package au.edu.unimelb.swen.oop.mokepon.driven.overworld;

public enum TileType {

    BARRIER('B'), FOREST('F'), GRASS('G'), OCEAN('O'), PATH('P'),
    ROCKS('R'), WATER('W'), EMPTY('\0'), UNKNOWN('?');

    private final char tileChar;

    TileType(char tileChar) {
        this.tileChar = tileChar;
    }

    public char getTileChar() {
        return tileChar;
    }

    public static TileType getTileType(char tileChar) {
        for (TileType tileType : values()) {
            if (tileType.tileChar == tileChar) return tileType;
        }
        return UNKNOWN;
    }

    public boolean isWalkable() {
        return this.tileChar != 'B' && this.tileChar != 'O' && this.tileChar != '\0'
                && this.tileChar != '?';
    }
}

package au.edu.unimelb.swen.oop.mokepon.driven.overworld;

import java.util.ArrayList;

public class WorldMap {

    private TileType[][] mapStatusMatrix;
    private int maxX;
    private int maxY;

    public WorldMap(ArrayList<ArrayList<Character>> worldMapListMatrix) {
        this.maxY = worldMapListMatrix.size();
        int maxX = 0;
        for (ArrayList<Character> row : worldMapListMatrix) {
            if (row.size() > maxX) maxX = row.size();
        }
        this.maxX = maxX;

        mapStatusMatrix = new TileType[maxY + 2][maxX + 2];

        for (int y = 0; y < maxY + 2; y++) {
            for (int x = 0; x < maxX + 2; x++) {
                mapStatusMatrix[y][x] = TileType.EMPTY;
            }
        }

        for (int y = 0; y < maxY; y++) {
            ArrayList<Character> row = worldMapListMatrix.get(y);
            for (int x = 0; x < row.size(); x++) {
                mapStatusMatrix[y + 1][x + 1] = getTileType(row.get(x));
            }
        }
    }

    private TileType getTileType(char c) {
        for (TileType tileType : TileType.values()) {
            if (tileType.getTileChar() == c) return tileType;
        }
        return TileType.EMPTY;
    }

    public TileType getTileType(Coordinate coordinate) {
        int realX = coordinate.getX();
        int realY = coordinate.getY();

        int mappedX = realX + 1;
        int mappedY = realY + 1;

        if (mappedX < 0 || mappedX >= maxX + 2 || mappedY < 0 || mappedY >= maxY + 2) {
            return TileType.EMPTY;
        }
        return mapStatusMatrix[mappedY][mappedX];
    }

    public int getMaxX() {
        return maxX;
    }
    public int getMaxY() {
        return maxY;
    }
}

package au.edu.unimelb.swen.oop.mokepon.event;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.function.Consumer;

public class EventBus {

    private HashMap<Class<?>, ArrayList<Consumer<?>>> subscriberListMap = new HashMap<>();

    public <T> void subscribe(Class<T> eventType, Consumer<T> handler) {
        subscriberListMap.computeIfAbsent(eventType, _ -> new ArrayList<>()).add(handler);
    }

    @SuppressWarnings("unchecked")
    public <T> void publish(T event) {
        ArrayList<Consumer<?>> handlerList = subscriberListMap.get(event.getClass());
        if (handlerList == null) return;

        handlerList.forEach(handler -> {
            ((Consumer<T>)handler).accept(event);
        });
    }

}

                fxBattleCardView, fxBattleItemView, fxBattleSkillView, fxBattleNameView
        );

        BattleNameView fxBattleNameView2 = new FxBattleNameView();
        BattleEnemyView fxBattleEnemyView = new FxBattleEnemyView(
                fxBattleCardView2, fxBattleNameView2
        );

        BattleLogView fxBattleLogView = new FxBattleLogView();

        MokeponBattleView mokeponBattleView = new MokeponBattleView(
                fxBattleSelfView,
                fxBattleEnemyView,
                fxBattleLogView,
                null,
                null
        );

        BattleService battleService = new BattleService(randomSource);
        BattleContext battleContext = new BattleContext();
        MokeponBattleController mokeponBattleController = new MokeponBattleController(
                mokeponBattleView, gameState, battleService, eventBus, battleContext, randomSource, screenManager
        );

        screenManager.register(ScreenState.BATTLE, mokeponBattleController);
        mokeponBattleController.init();
        mokeponBattleController.hide();
        //===============================================================================================


        // test space
        screenManager.switchTo(ScreenState.SELECT);

        MokeponLoader mokeponLoader = new MokeponLoader(gameProperty);
        Mokepon a = mokeponLoader.load("Mothique");
        Mokepon e = mokeponLoader.load("Needrake");

        };
        eventBus.subscribe(ChooseMokeponEvent.class, chooseMokeponEventConsumer2);

        // overworld -> select event
        Consumer<ExitOverworldEvent> exitOverworldEventConsumer = event -> {
            event.gameState().resetForNewSelection();

        eventBus.subscribe(ExitOverworldEvent.class, exitOverworldEventConsumer);

            gameBgView.enable(0);

                mokeponBattleController.getRoot()

        primaryStage.setScene(scene);
        primaryStage.show();

    private Properties loadPropertiesFromArgument() throws IOException {
        String argument = getParameters().getRaw().getFirst();
        if (!argument.startsWith("data=")) {
            throw new IllegalArgumentException("Missing expected argument in the form data=relativeFilePath");
        }

        Properties properties = new Properties();
        try (var input = Files.newInputStream(Path.of(argument.substring("data=".length())))) {
            properties.load(input);
        }
        return properties;
    }

    public static void main(String[] args) {
        Application.launch(args);
    }

package au.edu.unimelb.swen.oop.mokepon.state;

import javafx.beans.property.IntegerProperty;
import javafx.beans.property.SimpleIntegerProperty;

import java.util.LinkedHashMap;

public class Inventory {

    private LinkedHashMap<ItemType, IntegerProperty> itemMap = new LinkedHashMap<>();


    public Inventory() {
        for (ItemType itemType : ItemType.realValues()) {
            if (itemType == ItemType.UNKNOWN) continue;
            itemMap.put(itemType, new SimpleIntegerProperty(0));
        }
    }


    public LinkedHashMap<ItemType, IntegerProperty> getItemMap() {
        return itemMap;
    }

    public void add(ItemType itemType) {
        IntegerProperty itemNum = itemMap.get(itemType);
        itemNum.setValue(itemNum.get() + 1);
    }

    public IntegerProperty itemIntegerProperty(ItemType itemType) {
        return itemMap.get(itemType);
    }

    public int getCount(ItemType itemType) {
        return itemMap.get(itemType).getValue();
    }

    public boolean use(ItemType itemType) {
        IntegerProperty itemNum = itemMap.get(itemType);
        if (itemNum.getValue() <= 0) return false;

        itemNum.setValue(itemNum.getValue() - 1);
        return true;
    }

    public void clear() {
        itemMap.values().forEach(item -> {
            item.setValue(0);
        });
    }

}

package au.edu.unimelb.swen.oop.mokepon.state;

import java.util.Arrays;

public enum ItemType {
    UNKNOWN(-1), HEALTH(0), COFFEE(1), POISON(2);

    private final int itemCode;

    private static final ItemType[] REAL_ITEMS = Arrays.stream(values()).filter(ItemType::isReal).toArray(ItemType[]::new);

    ItemType(int itemCode) {
        this.itemCode = itemCode;
    }

    public int getItemCode() {
        return itemCode;
    }

    public static ItemType getItemType(int itemCode) {
        for (ItemType itemType : values()) {
            if (itemType.itemCode == itemCode) return itemType;
        }
        return UNKNOWN;
    }


    public boolean isReal() {
        return this != UNKNOWN;
    }
    public static ItemType[] realValues() {
        return REAL_ITEMS;
    }
}

package au.edu.unimelb.swen.oop.mokepon.util;

import javafx.scene.text.Font;

import java.io.InputStream;
import java.util.HashMap;
import java.util.function.Function;

public class FontLoader {

    private static HashMap<Integer, Font> fontCacheMap = new HashMap<>();

    private final static String GOSTISCH_FONT_URL = "/assets/GrenzeGotisch-Black.ttf";


    public static Font getGostischFont(int size) {

        Function<Integer, Font> fontGenerator = fontSize -> {
            Font resultFont = new Font(fontSize);
            try (InputStream inputFile = FontLoader.class.getResourceAsStream(GOSTISCH_FONT_URL)) {
                resultFont = Font.loadFont(inputFile, fontSize);
            } catch (Exception e) {
                e.printStackTrace();
            }
            return resultFont;
        };

        return fontCacheMap.computeIfAbsent(size, fontGenerator);
    }

}

package au.edu.unimelb.swen.oop.mokepon.util.overworld;

import au.edu.unimelb.swen.oop.mokepon.data.GameData;
import au.edu.unimelb.swen.oop.mokepon.driven.overworld.Coordinate;
import au.edu.unimelb.swen.oop.mokepon.driven.overworld.EnemyType;
import au.edu.unimelb.swen.oop.mokepon.util.ImageLoader;
import javafx.scene.effect.Light;
import javafx.scene.effect.Lighting;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;
import javafx.scene.paint.Color;

import java.util.LinkedHashMap;
import java.util.Properties;

public class EnemyInfoLoader {

    private static final String PREFIX = "map.mokepon";


    private static String keyOf(EnemyType type, String field) {
        return PREFIX + type.name() + "." + field;
    }

    public static ImageView getEnemyImageView(EnemyType enemyType) {
        Properties props = GameData.getGameData();


        String imgPath = props.getProperty(keyOf(enemyType, "image"));
        Image enemyImg = ImageLoader.load(imgPath, 32, 32, true);

        String colorStr = props.getProperty(keyOf(enemyType, "colour"), "white");
        Color enemyColor = parseColor(colorStr);

        ImageView iv = new ImageView(enemyImg);
        if (enemyColor != null && !Color.WHITE.equals(enemyColor)) {
            Light.Distant light = new Light.Distant();
            light.setColor(enemyColor);
            light.setAzimuth(-90);

            Lighting lighting = new Lighting(light);
            lighting.setSurfaceScale(0);
            lighting.setDiffuseConstant(1.0);
            iv.setEffect(lighting);
        }
        return iv;
    }

    private static Color parseColor(String s) {
        if (s == null || s.isBlank()) return Color.WHITE;
        try {
            return Color.web(s.trim());
        } catch (IllegalArgumentException e) {
            System.err.println("Unknown colour: " + s + ", fallback to WHITE");
            return Color.WHITE;
        }
    }

    public static LinkedHashMap<Coordinate, EnemyType> getCoordinateEnemyMap() {
        Properties props = GameData.getGameData();
        LinkedHashMap<Coordinate, EnemyType> map = new LinkedHashMap<>();

        for (EnemyType enemyType : EnemyType.values()) {
            for (int i = 0; i < 4; i++) {
                String key = PREFIX + enemyType.name() + "." + i;
                String coordStr = props.getProperty(key);
                if (coordStr == null) break;
                map.put(CoordinateLoader.getCoordinate(coordStr), enemyType);
            }
        }
        return map;
    }
}

package au.edu.unimelb.swen.oop.mokepon.util;

import javafx.geometry.Rectangle2D;
import javafx.stage.Screen;

public class ScreenPosition {

    public static final Rectangle2D SCREENBOUNDS = Screen.getPrimary().getBounds();
    public static final double BGWIDTH = 1280;
    public static final double BGHEIGHT = 768;
    public static final double MENUBARHEIGHT = 15;
    public static final double WINDOWHEIGHT = 10;
    public static final double WINDOWWIDTH = 5;
    public static final double SCREENWIDTH = BGWIDTH + WINDOWWIDTH * 2;
    public static final double SCREENHEIGHT = BGHEIGHT + MENUBARHEIGHT + WINDOWHEIGHT * 2;
    public static final double SCREENLEFT = SCREENBOUNDS.getMinX();
    public static final double SCREENTOP = SCREENBOUNDS.getMinY();


    public static double relativeCenterX(double idealX, double lengthX)
    {
        return idealX - (lengthX / 2);
    }

    public static double relativeCenterY(double idealY, double lengthY)
    {
        return idealY - (lengthY / 2);
    }

    private ScreenPosition()
    {
        throw new UnsupportedOperationException("Utility class cannot be instantiated");
    }

}

package au.edu.unimelb.swen.oop.mokepon.util;

import au.edu.unimelb.swen.oop.mokepon.driven.overworld.TileType;
import javafx.scene.image.Image;

import java.util.HashMap;

public class TileImgLoader {

    private static HashMap<Character, Image> imgMap = new HashMap<>();

    private static final String urlPrefix = "assets/tile_";
    private static final String urlSuffix = ".png";

    public static Image getTileImg(char c) {
        TileType tileType = TileType.getTileType(c);
        if (tileType.getTileChar() == '\0' || tileType.getTileChar() == '?') return null;
        return imgMap.computeIfAbsent(c, _ -> ImageLoader.load(urlPrefix + tileType.name() + urlSuffix, 32, 32, true));
    }

}

package au.edu.unimelb.swen.oop.mokepon.view.fx.battle;

import au.edu.unimelb.swen.oop.mokepon.driven.model.Mokepon;
import au.edu.unimelb.swen.oop.mokepon.util.ImageLoader;
import au.edu.unimelb.swen.oop.mokepon.view.view_interface.battle.BattleCardView;
import javafx.beans.property.IntegerProperty;
import javafx.beans.property.SimpleIntegerProperty;
import javafx.geometry.Pos;
import javafx.scene.Node;
import javafx.scene.control.ProgressBar;
import javafx.scene.image.ImageView;
import javafx.scene.layout.StackPane;
import javafx.scene.layout.VBox;
import javafx.scene.text.Text;

public class FxBattleCardView implements BattleCardView {


    private VBox battleCardRoot = new VBox(4);
    private ImageView selfImgView = new ImageView();
    private StackPane[] selfBarArr = new StackPane[2];
    private Text[] selfBarTxtArr = new Text[2];
    private ProgressBar[] selfProgressBarArr = new ProgressBar[2];

    private int maxHp = 0;
    private int maxEnergy = 0;

    private IntegerProperty curHp = new SimpleIntegerProperty(1);
    private IntegerProperty curEnergy = new SimpleIntegerProperty(1);

    private boolean textVisible = true;


    public FxBattleCardView() {

        for (int i = 0; i < selfBarArr.length; i++) {
            selfProgressBarArr[i] = new ProgressBar(1.0);
            selfBarTxtArr[i] = new Text("- / 0");
            selfBarTxtArr[i].setStyle("-fx-font-size: 16");
            selfBarArr[i] = new StackPane(selfProgressBarArr[i], selfBarTxtArr[i]);
        }

        selfProgressBarArr[0].setStyle("-fx-pref-width: 250;" +
                "-fx-pref-height: 27;" +
                "-fx-accent: #31c831;");

        selfProgressBarArr[1].setStyle("-fx-pref-width: 250;" +
                "-fx-pref-height: 27;" +
                "-fx-accent: #3e64d8;");

        curHp.addListener((observable, oldValue, newValue) -> {
            if (textVisible) selfBarTxtArr[0].setText(newValue + " / " + maxHp);
            selfProgressBarArr[0].setProgress(newValue.doubleValue() / maxHp);
        });

        curEnergy.addListener((observable, oldValue, newValue) -> {
            if (textVisible) selfBarTxtArr[1].setText(newValue + " / " + maxEnergy);
            selfProgressBarArr[1].setProgress(newValue.doubleValue() / maxEnergy);
        });


        battleCardRoot.getChildren().add(selfImgView);
        battleCardRoot.getChildren().addAll(selfBarArr);
        battleCardRoot.setAlignment(Pos.CENTER);
    }



    @Override
    public Node node() {
        return battleCardRoot;
    }

    @Override
    public void show(Mokepon mokepon) {
        selfImgView.setImage(ImageLoader.load(mokepon.getImgURL(), 256, 256));
        maxHp = mokepon.getMaxHP();
        maxEnergy = mokepon.getMaxEnergy();
    }

    @Override
    public void updateHP(int current) {
        curHp.setValue(current);
    }

    @Override
    public void updateEnergy(int current) {
        curEnergy.setValue(current);
    }

    @Override
    public void setTextVisible(boolean visible) {
        textVisible = visible;
        if (!visible) {
            for (Text txt : selfBarTxtArr) txt.setText("");
        }
    }
}

package au.edu.unimelb.swen.oop.mokepon.view.fx.battle.self;

import au.edu.unimelb.swen.oop.mokepon.view.view_interface.battle.self.BattleItemView;
import javafx.beans.property.BooleanProperty;
import javafx.beans.property.IntegerProperty;
import javafx.beans.property.SimpleBooleanProperty;
import javafx.geometry.Pos;
import javafx.scene.Node;
import javafx.scene.control.Button;
import javafx.scene.control.ContentDisplay;
import javafx.scene.layout.HBox;
import javafx.scene.text.Font;

import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.function.Consumer;

public class FxBattleItemView implements BattleItemView {


    private HBox battleItemRoot = new HBox(10);
    private LinkedHashMap<String, Button> itemBtnMap = new LinkedHashMap<>();

    private BooleanProperty globalDisableProperty = new SimpleBooleanProperty(false);


    public FxBattleItemView() {
        for (ItemType itemType : ItemType.realValues()) {
            Button itemBtn = new Button("0", ItemImgViewLoader.getItemImgView(itemType));
            itemBtn.setContentDisplay(ContentDisplay.TOP);
            itemBtn.setGraphicTextGap(10);
            itemBtn.setOpacity(0.2);
            itemBtn.setStyle("-fx-pref-width: 50;" +
                    "-fx-pref-height: 85;" +
                    "-fx-background-color: #d3d3d3;" +
                    "-fx-background-radius: 30px;" +
                    "-fx-text-fill: #000000");
            itemBtn.setFont(FontLoader.getGostischFont(24));
            itemBtnMap.put(itemType.name(), itemBtn);
        }
        battleItemRoot.getChildren().addAll(itemBtnMap.values());
        battleItemRoot.setAlignment(Pos.CENTER);
    }




    @Override
    public Node node() {
        return battleItemRoot;
    }

    @Override
    public void setItems(HashMap<ItemType, IntegerProperty> itemDeposit, Consumer<ItemType> onUseItem) {
        itemDeposit.forEach((itemType, integerProperty) -> {
            Button itemBtn = itemBtnMap.get(itemType.name());
            itemBtn.setOnMouseEntered(enter -> {
                itemBtn.setOpacity(0.5);
            });
            itemBtn.setOnMouseExited(exit -> {
                itemBtn.setOpacity(0.2);
            });
            itemBtn.setOnMouseClicked(event -> {
                itemBtn.setOpacity(0.75);
                onUseItem.accept(itemType);
            });

            itemBtn.disableProperty().bind(
                    integerProperty.lessThanOrEqualTo(0).or(globalDisableProperty)
            );
            itemBtn.textProperty().bind(integerProperty.asString());
        });
    }

    @Override
    public void setDisable(boolean disable) {
        globalDisableProperty.set(disable);
    }
}

package au.edu.unimelb.swen.oop.mokepon.view.fx.overworld;

import au.edu.unimelb.swen.oop.mokepon.driven.overworld.Coordinate;
import au.edu.unimelb.swen.oop.mokepon.util.TileImgLoader;
import au.edu.unimelb.swen.oop.mokepon.util.overworld.PlayerInfoLoader;
import au.edu.unimelb.swen.oop.mokepon.view.view_interface.overworld.TileView;
import javafx.geometry.Pos;
import javafx.scene.Node;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;
import javafx.scene.layout.ColumnConstraints;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.RowConstraints;
import javafx.scene.layout.StackPane;

import java.util.ArrayList;
import java.util.HashMap;

public class FxTileView implements TileView {

    private static final double TILE_SIZE = 32;

    private final StackPane wholeTileRoot = new StackPane();
    private final GridPane tileRoot = new GridPane();
    private final GridPane entityRoot = new GridPane();
    private final GridPane playerRoot = new GridPane();

    private final HashMap<Coordinate, ImageView> entityViewMap = new HashMap<>();
    private ImageView playerImgView = PlayerInfoLoader.getImgView();

    public FxTileView() {
        wholeTileRoot.getChildren().addAll(tileRoot, entityRoot, playerRoot);
        StackPane.setAlignment(tileRoot, Pos.TOP_LEFT);
        StackPane.setAlignment(entityRoot, Pos.TOP_LEFT);
        StackPane.setAlignment(playerRoot, Pos.TOP_LEFT);
    }

    @Override
    public Node node() {
        return wholeTileRoot;
    }

    @Override
    public void setTile(ArrayList<ArrayList<Character>> tileCharMatrix) {
        tileRoot.getChildren().clear();
        entityRoot.getChildren().clear();
        playerRoot.getChildren().clear();
        entityViewMap.clear();

        int rows = tileCharMatrix.size();
        int cols = rows == 0 ? 0 : tileCharMatrix.get(0).size();

        setupGridConstraints(tileRoot, cols, rows);
        setupGridConstraints(entityRoot, cols, rows);
        setupGridConstraints(playerRoot, cols, rows);

        for (int y = 0; y < rows; y++) {
            ArrayList<Character> row = tileCharMatrix.get(y);
            for (int x = 0; x < row.size(); x++) {
                Image tileImg = TileImgLoader.getTileImg(row.get(x));
                ImageView tileImgView = new ImageView(tileImg);
                tileRoot.add(tileImgView, x, y);
            }
        }

        playerImgView.setVisible(false);
        playerRoot.add(playerImgView, 0, 0);
    }

    private void setupGridConstraints(GridPane grid, int cols, int rows) {
        grid.getColumnConstraints().clear();
        grid.getRowConstraints().clear();

        for (int c = 0; c < cols; c++) {
            ColumnConstraints cc = new ColumnConstraints(TILE_SIZE);
            cc.setMinWidth(TILE_SIZE);
            cc.setPrefWidth(TILE_SIZE);
            cc.setMaxWidth(TILE_SIZE);
            grid.getColumnConstraints().add(cc);
        }
        for (int r = 0; r < rows; r++) {
            RowConstraints rc = new RowConstraints(TILE_SIZE);
            rc.setMinHeight(TILE_SIZE);
            rc.setPrefHeight(TILE_SIZE);
            rc.setMaxHeight(TILE_SIZE);
            grid.getRowConstraints().add(rc);
        }
    }

    @Override
    public void putEntity(Coordinate coordinate, ImageView entityImgView) {
        popEntity(coordinate);
        entityViewMap.put(coordinate, entityImgView);
        entityRoot.add(entityImgView, coordinate.getX(), coordinate.getY());
    }

    @Override
    public void popEntity(Coordinate coordinate) {
        ImageView entityImgView = entityViewMap.remove(coordinate);
        if (entityImgView != null) {
            entityRoot.getChildren().remove(entityImgView);
        }
    }

    @Override
    public void putPlayer(Coordinate coordinate) {
        playerImgView.setVisible(true);
        GridPane.setColumnIndex(playerImgView, coordinate.getX());
        GridPane.setRowIndex(playerImgView, coordinate.getY());
    }

    @Override
    public void popPlayer() {
        playerImgView.setVisible(false);
    }

    @Override
    public void setVisible(boolean visible) {
        wholeTileRoot.setVisible(visible);
    }

    @Override
    public void clear() {
        tileRoot.getChildren().clear();
        entityRoot.getChildren().clear();
        entityViewMap.clear();
        playerImgView.setVisible(false);
    }
}

    public Node node() {

    @Override
    public void show(Mokepon mokepon) {

            clear();
        } else {
            setMokeponImgView(ImageLoader.load("/assets/" + mokepon.getName() + ".png", 256, 256));
            setMokeponName(mokepon.getName());
            setMokeponTypeImgView(mokepon.getMokeponType());
            attrTxtList.get(0).setText(mokepon.getMaxHP() + "");
            attrTxtList.get(1).setText(mokepon.getMaxEnergy() + "");
            attrTxtList.get(2).setText(mokepon.getBaseStrength() + "");
            attrTxtList.get(3).setText(mokepon.getBaseAccuracy() + "");
            attrTxtList.get(4).setText(mokepon.getBaseDefence() + "");
            attrTxtList.get(5).setText(mokepon.getBaseDodge() + "");
            attrTxtList.get(6).setText(String.join(", ", mokepon.getSkills()));
            attrTxtList.get(7).setText(mokepon.getMokeponType() + "");

        setMokeponName(null);
        setMokeponTypeImgView(null);

            text.setText("");

package au.edu.unimelb.swen.oop.mokepon.view.fx.select;

import javafx.scene.layout.VBox;
import javafx.scene.text.Font;
import javafx.scene.text.Text;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.function.Consumer;

public class FxMokeponListView implements MokeponListView {

    private Text mokeponSelectTitle = new Text("Choose your Mokepon");

    private ScrollPane scrollPane = new ScrollPane();
    private VBox listBox = new VBox(10);

    private VBox selectBlock = new VBox(20);

    private HashMap<Mokepon, Button> buttonMap = new LinkedHashMap<>();
    private Font btnFont = FontLoader.getGostischFont(20);
    private Font titleFont = FontLoader.getGostischFont(34);
    private Button selectedButton;


    public FxMokeponListView() {
        listBox.setAlignment(Pos.CENTER);
        scrollPane.setContent(listBox);
        scrollPane.setFitToWidth(true);
        scrollPane.setVbarPolicy(ScrollPane.ScrollBarPolicy.AS_NEEDED);
        scrollPane.setHbarPolicy(ScrollPane.ScrollBarPolicy.NEVER);
        scrollPane.setBackground(Background.EMPTY);
        scrollPane.setStyle("-fx-background-color: transparent;");
        scrollPane.skinProperty().addListener((observer, oldSkin, newSkin) -> {
            Node viewport = scrollPane.lookup(".viewport");
            if (viewport != null)
                viewport.setStyle("-fx-background-color: transparent;");

            Node scrollBar = scrollPane.lookup(".scroll-bar");
            if (scrollBar != null)
                scrollBar.setStyle("-fx-background-color: transparent;");
        });
        scrollPane.setPrefSize(300, 600);

        mokeponSelectTitle.setFont(titleFont);
        mokeponSelectTitle.setStyle("-fx-fill: #000000;" +
                "-fx-stroke: #ffffff;" +
                "-fx-stroke-width: 2px;");

        selectBlock.getChildren().addAll(mokeponSelectTitle, scrollPane);
        selectBlock.setAlignment(Pos.CENTER);
    }


    @Override
    public Node node() {
        return selectBlock;
    }

    @Override
    public void setItems(ArrayList<Mokepon> mokeponList, Consumer<Mokepon> onSelect) {
        listBox.getChildren().clear();
        buttonMap.clear();
        for (Mokepon mokepon : mokeponList) {
            Button nameBtn = new Button(mokepon.getName());
            nameBtn.setFont(btnFont);
            nameBtn.setStyle("-fx-background-radius: 999px;" +
                    "-fx-background-color: #d3d3d3;" +
                    "-fx-text-fill: #000000;"
            );
            nameBtn.setOpacity(0.2);
            nameBtn.setOnMouseEntered(click -> {
                nameBtn.setOpacity(0.5);
            });
            nameBtn.setOnMouseExited(click -> {
                nameBtn.setOpacity(0.2);
            });
            nameBtn.setOnMouseClicked(click -> {
                nameBtn.setOpacity(0.75);
            });
            nameBtn.setPrefSize(280, 25);
            nameBtn.setOnAction(event -> onSelect.accept(mokepon));
            buttonMap.put(mokepon, nameBtn);
            listBox.getChildren().add(nameBtn);
        }
    }

    @Override
    public String getSelectedButtonName() {
        return selectedButton.getText();
    }

    @Override
    public void select(Mokepon mokepon) {
        clearSelection();
        selectedButton = buttonMap.get(mokepon);
    }

    @Override
    public void clearSelection() {
        selectedButton = null;
    }
}
